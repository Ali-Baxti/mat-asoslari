cmake_minimum_required(VERSION 3.16)
project(prime_checker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(prime_checker main.cpp)
add_executable(prime_benchmark benchmark.cpp)

enable_testing()
add_executable(prime_checker_tests tests.cpp)
add_test(NAME prime_checker_tests COMMAND prime_checker_tests)

option(BUILD_QT_GUI "Build the optional Qt 6 graphical interface" ON)
if(BUILD_QT_GUI)
    find_package(Qt6 COMPONENTS Widgets QUIET)
    if(Qt6_FOUND)
        add_executable(prime_checker_gui gui/main.cpp)
        target_link_libraries(prime_checker_gui PRIVATE Qt6::Widgets)
        target_include_directories(prime_checker_gui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        message(STATUS "Qt 6 Widgets found: GUI target enabled")
    else()
        message(STATUS "Qt 6 Widgets not found: skipping optional GUI target")
    endif()
endif()
